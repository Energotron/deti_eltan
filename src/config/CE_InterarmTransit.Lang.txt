            }
            03=else
            04 ^{
                01=unknown cePortalReady=ImportedFunction('CESecondMapAdapter','CEAdapterPortalReady');

                    18=unknown ceDestinationStar=ceOriginStar;
                    19=unknown ceBestDistSq=-1;
                    20=dword ceScanIndex=0;
                    21=while(ceScanIndex<GalaxyStars())
                    22 ^{
                        01=unknown ceCandidate=GalaxyStar(ceScanIndex);
                        02=if(StarToCon(ceCandidate)!=ceOriginSector)
                        03 ^{
                            01=unknown ceDx=CoordX(ceCandidate)-ceOriginX;
                            02=unknown ceDy=CoordY(ceCandidate)-ceOriginY;
                            03=unknown ceDistSq=ceDx*ceDx+ceDy*ceDy;
                            04=if(ceDistSq>ceBestDistSq)
                            05 ^{
                                01=ceBestDistSq=ceDistSq;
                                02=ceDestinationStar=ceCandidate;
                            }
                        }
                        04=ceScanIndex=ceScanIndex+1;
                    }
                    23=if(ceDestinationStar==ceOriginStar)
                    24 ^{
                        01=MessageBox(CT('CE.Transit.NoDestination'));
                    }
                    25=else
                    26 ^{
                        01=dword ceHole=HoleCreate2(ceOriginStar,ceDestinationStar);
                        02=if(ceHole)
                        03 ^{
                            01=HoleX1(ceHole,CoordX(Player())+350);
                            02=HoleY1(ceHole,CoordY(Player()));
                            03=HoleMap(ceHole,'SkipAB');
                            04=unknown ceRegisterPortal=ImportedFunction('CESecondMapAdapter','CEAdapterRegisterPortal');
                            05=if(!ceRegisterPortal(GalaxyPtr(),Id(ceHole))) MessageBox(CT('CE.Transit.PortalRegistrationFailed'));

                            07=ceArmCameraRecenterHole();
                        }
                    }
                }
            }
        }
    }
